71typedef struct { double x, y, z; } V3;
73static V3 V(double x, double y, double z) { V3 r = {x, y, z}; return r; }
75static V3 add(V3 a, V3 b) { return V(a.x + b.x, a.y + b.y, a.z + b.z); }
77static V3 sub(V3 a, V3 b) { return V(a.x - b.x, a.y - b.y, a.z - b.z); }
79static V3 scl(V3 a, double s) { return V(a.x * s, a.y * s, a.z * s); }
81static V3 mul(V3 a, V3 b) { return V(a.x * b.x, a.y * b.y, a.z * b.z); }
83static double dot(V3 a, V3 b) { return a.x * b.x + a.y * b.y + a.z * b.z; }
85static V3 cross(V3 a, V3 b) { return V(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x); }
87static V3 norm(V3 a) { return scl(a, 1.0 / sqrt(dot(a, a))); }
89static V3 neg(V3 a) { return V(-a.x, -a.y, -a.z); }
91static V3 mix(V3 a, V3 b, double t) { return add(scl(a, 1.0 - t), scl(b, t)); }

167static int hit_box(V3 o, V3 d, V3 lo, V3 hi, double tmax, double *t, V3 *n) {
168 double oa[3] = {o.x, o.y, o.z}, da[3] = {d.x, d.y, d.z};
169 double la[3] = {lo.x, lo.y, lo.z}, ha[3] = {hi.x, hi.y, hi.z};
170 double tn = -INF, tf = INF, sg = 0;
171 int ax = -1;
172 /* tn/tf: latest entry and earliest exit over the three slabs; ax remembers the entry axis */
173 for (int i = 0; i < 3; i++) {
174 if (fabs(da[i]) < 1e-12) {
175 if (oa[i] < la[i] || oa[i] > ha[i]) return 0;
176 continue;
177 }
178 double inv = 1.0 / da[i];
179 double t1 = (la[i] - oa[i]) * inv, t2 = (ha[i] - oa[i]) * inv;
180 double near_t = t1 < t2 ? t1 : t2, far_t = t1 < t2 ? t2 : t1;
181 if (near_t > tn) { tn = near_t; ax = i; sg = (t1 < t2) ? -1.0 : 1.0; }
182 if (far_t < tf) tf = far_t;
183 }
184 if (ax < 0 || tn > tf || tn < EPS || tn >= tmax) return 0;
185 double nn[3] = {0, 0, 0};
186 nn[ax] = sg;
187 *t = tn;
188 *n = V(nn[0], nn[1], nn[2]);
189 return 1;
190}

201static int hit_sphere(V3 o, V3 d, V3 c, double r, double tmax, double *t, V3 *n) {
202 V3 oc = sub(o, c);
203 double b = dot(oc, d), cc = dot(oc, oc) - r * r;
204 double disc = b * b - cc;
205 if (disc < 0) return 0;
206 double s = sqrt(disc);
207 double tt = -b - s;
208 if (tt < EPS) tt = -b + s;
209 if (tt < EPS || tt >= tmax) return 0;
210 *t = tt;
211 *n = norm(sub(add(o, scl(d, tt)), c));
212 return 1;
213}

225static int hit_cyl(V3 o, V3 d, V3 c, double y0, double y1, double r, double tmax, double *t, V3 *n) {
226 double best = tmax, ox = o.x - c.x, oz = o.z - c.z;
227 int hit = 0;
228 V3 bn = V(0, 0, 0);
229 /* side: (ox + t dx)^2 + (oz + t dz)^2 = r^2 */
230 double A = d.x * d.x + d.z * d.z;
231 if (A > 1e-12) {
232 double B = 2.0 * (ox * d.x + oz * d.z), C = ox * ox + oz * oz - r * r;
233 double disc = B * B - 4.0 * A * C;
234 if (disc >= 0) {
235 double s = sqrt(disc);
236 for (int k = 0; k < 2; k++) {
237 double tt = (-B + (k ? s : -s)) / (2.0 * A);
238 if (tt < EPS || tt >= best) continue;
239 double y = o.y + tt * d.y;
240 if (y < y0 || y > y1) continue;
241 best = tt;
242 bn = norm(V(ox + tt * d.x, 0, oz + tt * d.z));
243 hit = 1;
244 }
245 }
246 }
247 /* caps: planes y = y0 and y = y1, accepted inside the radius */
248 if (fabs(d.y) > 1e-12) {
249 for (int k = 0; k < 2; k++) {
250 double yc = k ? y1 : y0;
251 double tt = (yc - o.y) / d.y;
252 if (tt < EPS || tt >= best) continue;
253 double px = ox + tt * d.x, pz = oz + tt * d.z;
254 if (px * px + pz * pz > r * r) continue;
255 best = tt;
256 bn = V(0, k ? 1.0 : -1.0, 0);
257 hit = 1;
258 }
259 }
260 if (hit) { *t = best; *n = bn; }
261 return hit;
262}

275static int hit_cone(V3 o, V3 d, V3 c, double y0, double y1, double r, double tmax, double *t, V3 *n) {
276 double k = r / (y1 - y0), k2 = k * k;
277 double best = tmax, ox = o.x - c.x, oz = o.z - c.z, a = y1 - o.y;
278 int hit = 0;
279 V3 bn = V(0, 0, 0);
280 double A = d.x * d.x + d.z * d.z - k2 * d.y * d.y;
281 double B = 2.0 * (ox * d.x + oz * d.z + k2 * a * d.y);
282 double C = ox * ox + oz * oz - k2 * a * a;
283 double roots[2];
284 int nr = 0;
285 if (fabs(A) > 1e-12) {
286 double disc = B * B - 4.0 * A * C;
287 if (disc >= 0) {
288 double s = sqrt(disc);
289 roots[nr++] = (-B - s) / (2.0 * A);
290 roots[nr++] = (-B + s) / (2.0 * A);
291 }
292 } else if (fabs(B) > 1e-12) {
293 roots[nr++] = -C / B;
294 }
295 for (int i = 0; i < nr; i++) {
296 double tt = roots[i];
297 if (tt < EPS || tt >= best) continue;
298 double y = o.y + tt * d.y;
299 if (y < y0 || y > y1) continue;
300 double px = ox + tt * d.x, pz = oz + tt * d.z;
301 best = tt;
302 bn = norm(V(px, k2 * (y1 - y), pz));
303 hit = 1;
304 }
305 /* base disc at y = y0 */
306 if (fabs(d.y) > 1e-12) {
307 double tt = (y0 - o.y) / d.y;
308 if (tt >= EPS && tt < best) {
309 double px = ox + tt * d.x, pz = oz + tt * d.z;
310 if (px * px + pz * pz <= r * r) { best = tt; bn = V(0, -1, 0); hit = 1; }
311 }
312 }
313 if (hit) { *t = best; *n = bn; }
314 return hit;
315}

321static int hit_obj(const Obj *ob, V3 o, V3 d, double tmax, double *t, V3 *n) {
322 switch (ob->type) {
323 case BOX: return hit_box(o, d, ob->lo, ob->hi, tmax, t, n);
324 case SPHERE: return hit_sphere(o, d, ob->lo, ob->r, tmax, t, n);
325 case CYL: return hit_cyl(o, d, ob->lo, ob->lo.y, ob->hi.y, ob->r, tmax, t, n);
326 default: return hit_cone(o, d, ob->lo, ob->lo.y, ob->hi.y, ob->r, tmax, t, n);
327 }

330/* ---------- 2D grid over the XZ plane (Amanatides and Woo, 1987) ----------
331 * Each object is listed in every grid cell its XZ footprint touches. Objects with a huge
332 * footprint (the ground) are tested for every ray. A ray walks the cells it passes through
333 * (2D DDA) and stops once the nearest hit found so far is closer than the next cell. */
334#define GRID_N 36
335#define GRID_CELL 4.0
336#define GRID_MIN -72.0
337#define CELL_CAP 96
338#define GLOBAL_EXTENT 20.0
339
340static int cell_n[GRID_N * GRID_N];
341static int cell_obj[GRID_N * GRID_N][CELL_CAP];
342static int glob[MAX_OBJ];
343static int nglob = 0;
344
346static int cell_of(double v) {
347 int k = (int)floor((v - GRID_MIN) / GRID_CELL);
348 if (k < 0) k = 0;
349 if (k >= GRID_N) k = GRID_N - 1;
350 return k;
351}
352
359static void build_grid(void) {
360 nglob = 0;
361 memset(cell_n, 0, sizeof cell_n);
362 for (int i = 0; i < nobj; i++) {
363 Obj *ob = &objs[i];
364 double x0, x1, z0, z1;
365 if (ob->type == BOX) {
366 x0 = ob->lo.x; x1 = ob->hi.x; z0 = ob->lo.z; z1 = ob->hi.z;
367 } else if (ob->type == SPHERE) {
368 x0 = ob->lo.x - ob->r; x1 = ob->lo.x + ob->r; z0 = ob->lo.z - ob->r; z1 = ob->lo.z + ob->r;
369 } else {
370 x0 = ob->lo.x - ob->r; x1 = ob->lo.x + ob->r; z0 = ob->lo.z - ob->r; z1 = ob->lo.z + ob->r;
371 }
372 if (x1 - x0 > GLOBAL_EXTENT || z1 - z0 > GLOBAL_EXTENT) {
373 glob[nglob++] = i;
374 continue;
375 }
376 for (int ix = cell_of(x0); ix <= cell_of(x1); ix++)
377 for (int iz = cell_of(z0); iz <= cell_of(z1); iz++) {
378 int c = ix * GRID_N + iz;
379 if (cell_n[c] >= CELL_CAP) { fprintf(stderr, "grid cell full\n"); exit(1); }
380 cell_obj[c][cell_n[c]++] = i;
381 }
382 }
383}

398static int traverse(V3 o, V3 d, double tmax, int any, double *tout, V3 *nout) {
399 double best = tmax;
400 int found = -1;
401 V3 bn = V(0, 0, 0);
402 nrays++;
403
404 for (int g = 0; g < nglob; g++) {
405 int i = glob[g];
406 if (any && objs[i].emis > 0) continue;
407 double tt;
408 V3 nn;
409 if (hit_obj(&objs[i], o, d, best, &tt, &nn)) {
410 if (any) return i;
411 best = tt; found = i; bn = nn;
412 }
413 }
414
415 /* global objects were tested above; now walk the grid */
416 /* where the ray enters and leaves the grid square (slab test in x and z) */
417 const double gmin = GRID_MIN, gmax = GRID_MIN + GRID_N * GRID_CELL;
418 double t0 = 0, gexit = INF;
419 double oa[2] = {o.x, o.z}, da[2] = {d.x, d.z};
420 for (int a = 0; a < 2; a++) {
421 if (fabs(da[a]) < 1e-12) {
422 if (oa[a] < gmin || oa[a] > gmax) goto finish;
423 continue;
424 }
425 double inv = 1.0 / da[a];
426 double ta = (gmin - oa[a]) * inv, tb = (gmax - oa[a]) * inv;
427 if (ta > tb) { double s = ta; ta = tb; tb = s; }
428 if (ta > t0) t0 = ta;
429 if (tb < gexit) gexit = tb;
430 if (t0 > gexit) goto finish;
431 }
432 if (gexit > best) gexit = best;
433
434 {
435 /* start cell, step direction (sx, sz), tDelta (tdx, tdz) and tMax (tmx, tmz) as in the paper */
436 double px = o.x + d.x * t0, pz = o.z + d.z * t0;
437 int ix = cell_of(px), iz = cell_of(pz);
438 int sx = d.x > 0 ? 1 : -1, sz = d.z > 0 ? 1 : -1;
439 double tdx = fabs(d.x) > 1e-12 ? GRID_CELL / fabs(d.x) : INF;
440 double tdz = fabs(d.z) > 1e-12 ? GRID_CELL / fabs(d.z) : INF;
441 double bx = sx > 0 ? GRID_MIN + (ix + 1) * GRID_CELL : GRID_MIN + ix * GRID_CELL;
442 double bz = sz > 0 ? GRID_MIN + (iz + 1) * GRID_CELL : GRID_MIN + iz * GRID_CELL;
443 double tmx = fabs(d.x) > 1e-12 ? t0 + (bx - px) / d.x : INF;
444 double tmz = fabs(d.z) > 1e-12 ? t0 + (bz - pz) / d.z : INF;
445
446 for (;;) {
447 int c = ix * GRID_N + iz;
448 for (int k = 0; k < cell_n[c]; k++) {
449 int i = cell_obj[c][k];
450 if (any && objs[i].emis > 0) continue;
451 double tt;
452 V3 nn;
453 if (hit_obj(&objs[i], o, d, best, &tt, &nn)) {
454 if (any) return i;
455 best = tt; found = i; bn = nn;
456 }
457 }
458 double tex = tmx < tmz ? tmx : tmz;
459 if (tex >= best || tex > gexit) break;
460 if (tmx < tmz) { ix += sx; tmx += tdx; }
461 else { iz += sz; tmz += tdz; }
462 if (ix < 0 || ix >= GRID_N || iz < 0 || iz >= GRID_N) break;
463 }
464 }
465
466finish:
467 if (any) return -1;
468 if (found < 0) return -1;
469 *tout = best;
470 *nout = bn;
471 return found;
472}

480static int closest(V3 o, V3 d, double *t, V3 *n, int *idx) {
481 double tt;
482 V3 nn;
483 int i = traverse(o, d, INF, 0, &tt, &nn);
484 if (i < 0) return 0;
485 *t = tt; *n = nn; *idx = i;
486 return 1;

514static V3 trace(V3 o, V3 d, int depth) {
515 double t;
516 V3 n;
517 int idx;
518 if (!closest(o, d, &t, &n, &idx)) return sky(d);
519 Obj *ob = &objs[idx];
520 if (ob->emis > 0) return scl(ob->col, ob->emis);
521
522 V3 p = add(o, scl(d, t));
523 if (dot(n, d) > 0) n = neg(n);
524 V3 po = add(p, scl(n, 1e-3));
525
526 V3 col = mul(ob->col, ambient);
527
528 /* sun: directional, hard shadow, only when above the horizon */
529 double ndl = dot(n, sun_dir);
530 if (sun_dir.y > 0.02 && ndl > 0 && !occluded(po, sun_dir, INF))
531 col = add(col, scl(mul(ob->col, sun_col), sun_k * ndl));
532
533 /* lamp posts: point lights with distance falloff and hard shadows */
534 for (int i = 0; i < nlamp; i++) {
535 V3 to = sub(lamps[i].pos, p);
536 double dist = sqrt(dot(to, to));
537 V3 L = scl(to, 1.0 / dist);
538 double l = dot(n, L);
539 if (l <= 0) continue;
540 if (occluded(po, L, dist)) continue;
541 double att = 1.0 / (1.0 + 0.08 * dist * dist);
542 col = add(col, scl(mul(ob->col, lamps[i].col), lamps[i].k * att * l));
543 }
544
545 /* mirror reflection */
546 if (ob->refl > 0 && depth > 0) {
547 V3 r = sub(d, scl(n, 2.0 * dot(d, n)));
548 V3 rc = trace(po, norm(r), depth - 1);
549 col = add(scl(col, 1.0 - ob->refl), scl(rc, ob->refl));
550 }
551 return col;
552}

555#define SUN_ANGLE 1.0
562static void update_light(double a) {
563 sun_dir = norm(V(-0.7, sin(a), 0.35));
564 daylt = fmin(1.0, fmax(0.0, sun_dir.y * 3.0));
565 sun_k = fmax(0.0, sun_dir.y) * 1.2;
566 double warm = fmin(1.0, fmax(0.0, 1.0 - sun_dir.y * 2.5));
567 sun_col = V(1.0, 0.9 - 0.2 * warm, 0.75 - 0.3 * warm);
568 ambient = V(0.05 + 0.2 * daylt, 0.05 + 0.2 * daylt, 0.08 + 0.25 * daylt);
569
570 V3 hor = mix(V(0.03, 0.03, 0.07), V(0.7, 0.8, 0.92), daylt);
571 double dusk = fmax(0.0, 1.0 - fabs(sun_dir.y) * 4.0) * 0.8;
572 sky_hor = mix(hor, V(0.95, 0.55, 0.32), dusk);
573 sky_top = mix(V(0.0, 0.0, 0.03), V(0.25, 0.45, 0.9), daylt);
574
575 for (int i = 0; i < nlamp; i++) lamps[i].k = 1.2 + 2.5 * (1.0 - daylt);
576}

661static void render_row(Job *j, int y) {
662 int n = j->spp;
663 double aspect = (double)j->W / j->H;
664 for (int x = 0; x < j->W; x++) {
665 V3 acc = V(0, 0, 0);
666 for (int sy = 0; sy < n; sy++)
667 for (int sx = 0; sx < n; sx++) {
668 double px = (x + (sx + 0.5) / n) / j->W;
669 double py = (y + (sy + 0.5) / n) / j->H;
670 double u = (2.0 * px - 1.0) * aspect * j->vw.th;
671 double w = (1.0 - 2.0 * py) * j->vw.th;
672 V3 d = norm(add(j->vw.fwd, add(scl(j->vw.right, u), scl(j->vw.up, w))));
673 acc = add(acc, trace(j->vw.eye, d, MAX_DEPTH));
674 }
675 acc = scl(acc, 1.0 / (n * n));
676 unsigned char *p = j->img + 3 * (y * j->W + x);
677 double c[3] = {acc.x, acc.y, acc.z};
678 for (int i = 0; i < 3; i++) {
679 double v = c[i];
680 if (v < 0) v = 0;
681 if (v > 1) v = 1;
682 v = pow(v, 1.0 / 2.2);
683 p[i] = (unsigned char)(v * 255.0 + 0.5);
684 }
685 }
686}

689static void *worker(void *arg) {
690 Worker *w = (Worker *)arg;
691 Job *j = w->job;
692 nrays = 0;
693 for (;;) {
694 pthread_mutex_lock(&j->lock);
695 int y = j->next_row < j->H ? j->next_row++ : -1;
696 pthread_mutex_unlock(&j->lock);
697 if (y < 0) break;
698 render_row(j, y);
699 }
700 w->rays = nrays;
701 return NULL;
702}

709static void render_frame(unsigned char *img, int W, int H, View vw, int spp, int nthreads) {
710 Job j;
711 memset(&j, 0, sizeof j);
712 j.W = W; j.H = H; j.spp = spp; j.vw = vw; j.img = img; j.next_row = 0;
713 pthread_mutex_init(&j.lock, NULL);
714
715 pthread_t tid[MAX_THREADS];
716 Worker wk[MAX_THREADS];
717 for (int i = 0; i < nthreads; i++) {
718 wk[i].job = &j;
719 wk[i].rays = 0;
720 if (pthread_create(&tid[i], NULL, worker, &wk[i]) != 0) {
721 fprintf(stderr, "pthread_create failed\n");
722 exit(1);
723 }
724 }
725 for (int i = 0; i < nthreads; i++) pthread_join(tid[i], NULL);
726 pthread_mutex_destroy(&j.lock);
727}

769static int write_bmp(const char *path, int w, int h, const unsigned char *rgb) {
770 int row = (w * 3 + 3) & ~3;
771 unsigned int size = 54u + (unsigned int)row * (unsigned int)h;
772 unsigned char hdr[54];
773 memset(hdr, 0, sizeof hdr);
774 hdr[0] = 'B'; hdr[1] = 'M';
775 put32(hdr + 2, size); put32(hdr + 10, 54); put32(hdr + 14, 40);
776 put32(hdr + 18, (unsigned int)w); put32(hdr + 22, (unsigned int)h);
777 put16(hdr + 26, 1); put16(hdr + 28, 24);
778 put32(hdr + 34, (unsigned int)row * (unsigned int)h);
779 FILE *f = fopen(path, "wb");
780 if (!f) return 0;
781 fwrite(hdr, 1, 54, f);
782 unsigned char *line = calloc(row, 1);
783 for (int y = h - 1; y >= 0; y--) {
784 for (int x = 0; x < w; x++) {
785 const unsigned char *p = rgb + 3 * (y * w + x);
786 line[3 * x] = p[2]; line[3 * x + 1] = p[1]; line[3 * x + 2] = p[0];
787 }
788 fwrite(line, 1, row, f);
789 }
790 free(line);
791 fclose(f);
792 return 1;
793}

860static void handle_move(double dt) {
861 double move = WALK_SPEED * dt * (key_down(KEY_RUN) ? 3.0 : 1.0);
862 double yr = cam.yaw * PI / 180.0;
863 double fx = sin(yr), fz = cos(yr); /* ground-plane forward */
864 double rx = cos(yr), rz = -sin(yr); /* ground-plane right */
865 if (key_down(KEY_W)) { cam.x += fx * move; cam.z += fz * move; }
866 if (key_down(KEY_S)) { cam.x -= fx * move; cam.z -= fz * move; }
867 if (key_down(KEY_D)) { cam.x += rx * move; cam.z += rz * move; }
868 if (key_down(KEY_A)) { cam.x -= rx * move; cam.z -= rz * move; }
869 if (key_down(KEY_ZOOM_IN)) cam.fov = fmax(20, cam.fov - 30 * dt);
870 if (key_down(KEY_ZOOM_OUT)) cam.fov = fmin(100, cam.fov + 30 * dt);
